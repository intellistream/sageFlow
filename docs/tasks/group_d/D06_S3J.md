## 任务概述

实现 S3J (Scalable Streaming Similarity Join) 方法，基于 DEBS'23 论文，这是一个自适应的分布式流式相似度 Join 系统。

## 核心组件

### 1. AdaptivePartitioner

```cpp
#pragma once

#include "stream/partitioner/centroid_partitioner.h"
#include <chrono>
#include <atomic>

namespace sageFlow {

/**
 * @brief 自适应分区器
 * 
 * 根据数据分布和负载自动调整分区策略
 */
class AdaptivePartitioner : public CentroidPartitioner {
public:
    struct AdaptConfig {
        int initial_partitions = 16;
        int64_t adapt_interval_ms = 1000;
        double load_threshold = 0.3;
        double split_threshold = 2.0;   // 分裂阈值（相对均值）
        double merge_threshold = 0.3;   // 合并阈值（相对均值）
    };
    
    explicit AdaptivePartitioner(const AdaptConfig& config);
    
    /**
     * @brief 更新分区统计
     */
    void updateStats(int partition, int64_t latency_us, size_t data_size);
    
    /**
     * @brief 检查并执行自适应调整
     */
    void checkAndAdapt();
    
    /**
     * @brief 获取当前分区数
     */
    int getCurrentNumPartitions() const;
    
    /**
     * @brief 获取自适应历史
     */
    struct AdaptHistory {
        std::chrono::steady_clock::time_point timestamp;
        std::string action;  // "split", "merge", "rebalance"
        int partition_id;
    };
    std::vector<AdaptHistory> getHistory() const;

private:
    AdaptConfig adapt_config_;
    std::atomic<int64_t> last_adapt_time_;
    
    // 每分区统计
    struct PartitionStats {
        std::atomic<size_t> count{0};
        std::atomic<int64_t> total_latency_us{0};
        std::atomic<size_t> data_size{0};
    };
    std::vector<PartitionStats> partition_stats_;
    
    std::vector<AdaptHistory> history_;
    mutable std::mutex history_mutex_;
    
    /**
     * @brief 分裂过载分区
     */
    void splitPartition(int partition);
    
    /**
     * @brief 合并低负载分区
     */
    void mergePartitions(int partition1, int partition2);
    
    /**
     * @brief 计算负载不均衡度
     */
    double computeLoadImbalance() const;
};

} // namespace sageFlow
```

### 2. AdaptiveIndexSelector

```cpp
#pragma once

#include "index/index.h"
#include <memory>
#include <unordered_map>

namespace sageFlow {

/**
 * @brief 自适应索引选择器
 * 
 * 根据数据特性选择最佳索引类型
 */
class AdaptiveIndexSelector {
public:
    struct Config {
        double switch_threshold = 0.2;  // 性能差距阈值
        int sample_size = 100;          // 采样大小
        int eval_interval = 1000;       // 评估间隔（记录数）
    };
    
    explicit AdaptiveIndexSelector(const Config& config);
    
    /**
     * @brief 选择最佳索引类型
     */
    IndexType selectBestIndex(
        int dimension,
        size_t data_size,
        double query_rate) const;
    
    /**
     * @brief 评估索引性能
     */
    struct IndexPerformance {
        double avg_latency_us;
        double recall;
        double memory_mb;
    };
    IndexPerformance evaluateIndex(
        IndexType type,
        const std::vector<VectorRecord>& samples,
        const std::vector<VectorRecord>& queries) const;
    
    /**
     * @brief 获取推荐的索引配置
     */
    std::map<std::string, std::string> getRecommendedParams(
        IndexType type,
        int dimension,
        size_t expected_size) const;

private:
    Config config_;
    
    // 历史性能缓存
    mutable std::unordered_map<IndexType, IndexPerformance> perf_cache_;
};

} // namespace sageFlow
```

### 3. S3JMethod 主类

```cpp
#pragma once

#include "operator/join_operator_methods/base_method.h"
#include "operator/join_operator_methods/s3j_components/adaptive_partitioner.h"
#include "operator/join_operator_methods/s3j_components/adaptive_index_selector.h"

namespace sageFlow {

/**
 * @brief S3J 方法
 * 
 * DEBS'23 论文的实现：自适应分布式流式相似度 Join
 */
class S3JMethod : public BaseMethod {
public:
    struct Config {
        double similarity_threshold = 0.8;
        int num_partitions = 16;
        int64_t adapt_interval_ms = 1000;
        double load_threshold = 0.3;
        double index_switch_threshold = 0.2;
        bool enable_metrics = true;
    };
    
    explicit S3JMethod(const Config& config);
    ~S3JMethod() override = default;
    
    std::string getName() const override { return "S3J"; }
    JoinStrategyConfig getRecommendedConfig() const override;
    
    void open(const RuntimeContext& context, 
              JoinOperatorState* state) override;
    
    std::vector<std::unique_ptr<VectorRecord>> ExecuteEager(
        const VectorRecord& query, int slot) override;
    
    std::vector<std::unique_ptr<VectorRecord>> ExecuteLazy(
        const std::deque<std::unique_ptr<VectorRecord>>& queries,
        int slot) override;
    
    void close() override;
    
    /**
     * @brief 获取运行时指标
     */
    struct Metrics {
        double avg_latency_ms;
        double throughput_qps;
        double recall_estimate;
        int current_partitions;
        std::string current_index_type;
        std::vector<AdaptivePartitioner::AdaptHistory> adapt_history;
    };
    Metrics getMetrics() const;
    
    /**
     * @brief 强制触发自适应调整
     */
    void forceAdapt();

private:
    Config config_;
    JoinOperatorState* state_ = nullptr;
    size_t subtask_index_ = 0;
    
    std::shared_ptr<AdaptivePartitioner> partitioner_;
    std::shared_ptr<AdaptiveIndexSelector> index_selector_;
    ConcurrencyManager* concurrency_manager_ = nullptr;
    int32_t index_id_ = -1;
    IndexType current_index_type_;
    
    // 指标收集
    struct MetricsCollector {
        std::atomic<size_t> query_count{0};
        std::atomic<int64_t> total_latency_us{0};
        std::atomic<size_t> match_count{0};
        std::chrono::steady_clock::time_point start_time;
    };
    MetricsCollector metrics_;
    
    /**
     * @brief 执行自适应检查
     */
    void maybeAdapt();
    
    /**
     * @brief 切换索引类型
     */
    void switchIndex(IndexType new_type);
    
    /**
     * @brief 在分区内搜索
     */
    std::vector<std::shared_ptr<const VectorRecord>> searchInPartition(
        const VectorRecord& query, int partition, double threshold);
};

// 自动注册
REGISTER_JOIN_METHOD(S3JMethod, "s3j");

} // namespace sageFlow
```

## 自适应策略

### 分区自适应

```cpp
// 检查负载不均衡
double imbalance = computeLoadImbalance();
if (imbalance > config.load_threshold) {
    // 找到最大和最小分区
    int max_partition = findMaxLoadPartition();
    int min_partition = findMinLoadPartition();
    
    if (partition_load[max_partition] > avg_load * split_threshold) {
        splitPartition(max_partition);
    } else if (partition_load[min_partition] < avg_load * merge_threshold) {
        mergePartitions(min_partition, findNeighbor(min_partition));
    }
}
```

### 索引自适应

```cpp
// 定期评估索引性能
if (shouldEvaluateIndex()) {
    auto current_perf = evaluateCurrentIndex();
    auto alternative_perf = evaluateAlternativeIndex();
    
    if (alternative_perf.latency < current_perf.latency * (1 - switch_threshold)) {
        switchIndex(alternative_type);
    }
}
```
